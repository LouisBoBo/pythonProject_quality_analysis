# PCB品质分析处理系统设计

## 一、系统概述

### 1.1 系统目标
本系统旨在建立PCB生产企业的全面质量管理体系，实现质量数据的自动采集、智能分析、快速处理和持续改进，提升产品质量，降低质量成本。

### 1.2 核心功能
- 8D报告自动生成与管理
- SPC统计过程控制
- 不良品根本原因分析
- 质量趋势监控与预警
- 客户投诉处理跟踪
- 改善方案效果验证

### 1.3 系统特点
- **智能化分析**: 基于AI算法的质量预测
- **自动化处理**: 8D报告自动生成
- **实时监控**: 质量数据实时采集
- **可视化展示**: 数据看板直观展示
- **标准化流程**: 符合质量管理体系要求

### 4.1 SPC统计分析算法
```python
class SPCStatisticalAnalysis:
    def __init__(self, data_points, subgroup_size=5):
        self.data_points = data_points
        self.subgroup_size = subgroup_size
        self.subgroups = self._create_subgroups()
        
    def _create_subgroups(self):
        """创建子组"""
        subgroups = []
        for i in range(0, len(self.data_points), self.subgroup_size):
            subgroup = self.data_points[i:i+self.subgroup_size]
            if len(subgroup) == self.subgroup_size:
                subgroups.append(subgroup)
        return subgroups
    
    def calculate_control_limits(self, chart_type="Xbar-R"):
        """计算控制限"""
        if chart_type == "Xbar-R":
            return self._calculate_xbar_r_limits()
        elif chart_type == "Xbar-S":
            return self._calculate_xbar_s_limits()
        elif chart_type == "p-chart":
            return self._calculate_p_chart_limits()
        elif chart_type == "np-chart":
            return self._calculate_np_chart_limits()
        elif chart_type == "c-chart":
            return self._calculate_c_chart_limits()
        elif chart_type == "u-chart":
            return self._calculate_u_chart_limits()
        else:
            raise ValueError(f"不支持的图表类型: {chart_type}")
    
    def _calculate_xbar_r_limits(self):
        """计算Xbar-R图控制限"""
        # 计算每个子组的均值和极差
        subgroup_means = [np.mean(subgroup) for subgroup in self.subgroups]
        subgroup_ranges = [np.max(subgroup) - np.min(subgroup) for subgroup in self.subgroups]
        
        # 计算总均值和平均极差
        xbar_bar = np.mean(subgroup_means)
        r_bar = np.mean(subgroup_ranges)
        
        # 查表获取系数 (假设A2=0.577, D3=0, D4=2.114)
        A2 = 0.577
        D3 = 0
        D4 = 2.114
        
        # 计算控制限
        xbar_ucl = xbar_bar + A2 * r_bar
        xbar_cl = xbar_bar
        xbar_lcl = xbar_bar - A2 * r_bar
        
        r_ucl = D4 * r_bar
        r_cl = r_bar
        r_lcl = D3 * r_bar
        
        return {
            'xbar': {'UCL': xbar_ucl, 'CL': xbar_cl, 'LCL': xbar_lcl},
            'r': {'UCL': r_ucl, 'CL': r_cl, 'LCL': r_lcl},
            'subgroup_means': subgroup_means,
            'subgroup_ranges': subgroup_ranges
        }
    
    def calculate_process_capability(self, usl, lsl):
        """计算过程能力指数"""
        # 计算总体均值和标准差
        all_data = np.concatenate(self.subgroups)
        overall_mean = np.mean(all_data)
        overall_std = np.std(all_data, ddof=1)
        
        # 计算组内标准差
        subgroup_means = [np.mean(subgroup) for subgroup in self.subgroups]
        subgroup_stds = [np.std(subgroup, ddof=1) for subgroup in self.subgroups]
        within_std = np.mean(subgroup_stds) / self._get_c4_factor(self.subgroup_size)
        
        # 计算过程能力指数
        cp = (usl - lsl) / (6 * within_std) if within_std > 0 else 0
        cpu = (usl - overall_mean) / (3 * within_std) if within_std > 0 else 0
        cpl = (overall_mean - lsl) / (3 * within_std) if within_std > 0 else 0
        cpk = min(cpu, cpl)
        
        # 计算过程性能指数
        pp = (usl - lsl) / (6 * overall_std) if overall_std > 0 else 0
        ppu = (usl - overall_mean) / (3 * overall_std) if overall_std > 0 else 0
        ppl = (overall_mean - lsl) / (3 * overall_std) if overall_std > 0 else 0
        ppk = min(ppu, ppl)
        
        return {
            'cp': cp,
            'cpk': cpk,
            'pp': pp,
            'ppk': ppk,
            'overall_mean': overall_mean,
            'overall_std': overall_std,
            'within_std': within_std,
            'usl': usl,
            'lsl': lsl
        }
    
    def _get_c4_factor(self, n):
        """获取c4系数"""
        # 简化版，实际应查表
        c4_factors = {
            2: 0.7979,
            3: 0.8862,
            4: 0.9213,
            5: 0.9400,
            6: 0.9515,
            7: 0.9594,
            8: 0.9650,
            9: 0.9693,
            10: 0.9727
        }
        return c4_factors.get(n, 1.0)
```

```python
class DefectAnalysis:
    def __init__(self, defect_data, production_data):
        self.defect_data = defect_data
        self.production_data = production_data
        
    def analyze_defect_patterns(self):
        """分析缺陷模式"""
        # 按缺陷类型统计
        defect_by_type = defaultdict(int)
        defect_by_process = defaultdict(int)
        defect_by_equipment = defaultdict(int)
        defect_by_operator = defaultdict(int)
        
        for defect in self.defect_data:
            defect_type = defect.get('defect_type', '未知')
            process = defect.get('process_step', '未知')
            equipment = defect.get('equipment_id', '未知')
            operator = defect.get('operator_id', '未知')
            
            defect_by_type[defect_type] += defect.get('defect_quantity', 0)
            defect_by_process[process] += defect.get('defect_quantity', 0)
            defect_by_equipment[equipment] += defect.get('defect_quantity', 0)
            defect_by_operator[operator] += defect.get('defect_quantity', 0)
        
        # 计算缺陷率
        total_defects = sum(defect_by_type.values())
        total_production = sum(self.production_data.values())
        overall_defect_rate = total_defects / total_production if total_production > 0 else 0
        
        # 识别主要缺陷类型
        top_defect_types = sorted(defect_by_type.items(), key=lambda x: x[1], reverse=True)[:5]
        
        # 识别瓶颈工序
        top_defect_processes = sorted(defect_by_process.items(), key=lambda x: x[1], reverse=True)[:3]
        
        return {
            'overall_defect_rate': overall_defect_rate,
            'total_defects': total_defects,
            'total_production': total_production,
            'defect_by_type': dict(defect_by_type),
            'defect_by_process': dict(defect_by_process),
            'defect_by_equipment': dict(defect_by_equipment),
            'defect_by_operator': dict(defect_by_operator),
            'top_defect_types': top_defect_types,
            'top_defect_processes': top_defect_processes,
            'pareto_analysis': self._perform_pareto_analysis(defect_by_type)
        }
    
    def _perform_pareto_analysis(self, defect_data):
        """进行帕累托分析"""
        sorted_defects = sorted(defect_data.items(), key=lambda x: x[1], reverse=True)
        total_defects = sum(defect_data.values())
        
        pareto_data = []
        cumulative_percentage = 0
        
        for defect_type, count in sorted_defects:
            percentage = count / total_defects * 100
            cumulative_percentage += percentage
            
            pareto_data.append({
                'defect_type': defect_type,
                'count': count,
                'percentage': percentage,
                'cumulative_percentage': cumulative_percentage
            })
        
        # 识别关键少数（通常为80%问题由20%原因造成）
        critical_defects = []
        for item in pareto_data:
            if item['cumulative_percentage'] <= 80:
                critical_defects.append(item['defect_type'])
        
        return {
            'pareto_data': pareto_data,
            'critical_defects': critical_defects,
            'vital_few_count': len(critical_defects),
            'trivial_many_count': len(sorted_defects) - len(critical_defects)
        }
    
    def predict_defect_trends(self, historical_data, periods=30):
        """预测缺陷趋势"""
        # 时间序列分析
        dates = [item['date'] for item in historical_data]
        defect_rates = [item['defect_rate'] for item in historical_data]
        
        # 使用移动平均法
        moving_avg = self._calculate_moving_average(defect_rates, window=7)
        
        # 使用指数平滑法
        exp_smoothing = self._calculate_exponential_smoothing(defect_rates, alpha=0.3)
        
        # 趋势预测
        trend = self._calculate_trend(defect_rates)
        
        # 预测未来值
        future_predictions = []
        last_value = defect_rates[-1] if defect_rates else 0
        
        for i in range(periods):
            predicted = last_value * (1 + trend)
            future_predictions.append({
                'date': (datetime.now() + timedelta(days=i+1)).strftime('%Y-%m-%d'),
                'predicted_rate': predicted,
                'confidence_interval': [predicted * 0.9, predicted * 1.1]
            })
        
        return {
            'historical_data': historical_data,
            'moving_average': moving_avg,
            'exponential_smoothing': exp_smoothing,
            'trend': trend,
            'future_predictions': future_predictions,
            'alert_level': '高' if trend > 0.1 else ('中' if trend > 0.05 else '低')
        }
```
